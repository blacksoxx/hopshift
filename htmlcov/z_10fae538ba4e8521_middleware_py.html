1"""HTTP hardening: security headers and origin checks for cookie-authenticated mutations. 

2 

3Evidence for SOC 2 CC6.6 / CC6.7, HIPAA 164.312(e)(1) and CIS Controls 12/13. The security 

4workstream extends this module (rate limiting, HSTS tuning, CSP policy modes). 

5""" 

6from __future__ import annotations 

7 

8from typing import Any, Awaitable, Callable 

9from urllib.parse import urlsplit 

10 

11from starlette.middleware.base import BaseHTTPMiddleware 

12from starlette.requests import Request 

13from starlette.responses import PlainTextResponse, Response 

14 

15SESSION_COOKIE = "vma_session" 

16UNSAFE_METHODS = {"POST", "PUT", "PATCH", "DELETE"} 

17 

18STRICT_CSP = ( 

19 "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'; img-src 'self' data: blob:; " 

20 "font-src 'self' data:; connect-src 'self'; frame-ancestors 'none'; base-uri 'self'; form-action 'self'; " 

21 "object-src 'none'" 

22) 

23# Swagger UI and ReDoc load their assets from a CDN; they get a relaxed policy on their own paths only. 

24DOCS_CSP = ( 

25 "default-src 'self'; script-src 'self' https://cdn.jsdelivr.net 'unsafe-inline'; " 

26 "style-src 'self' https://cdn.jsdelivr.net 'unsafe-inline'; img-src 'self' data: https://fastapi.tiangolo.com; " 

27 "font-src 'self' data:; connect-src 'self'; frame-ancestors 'none'; base-uri 'self'; object-src 'none'" 

28) 

29DOCS_PATHS = ("/docs", "/redoc", "/openapi.json") 

30 

31 

32def is_https(request: Request) -> bool: 

33 return request.url.scheme == "https" or request.headers.get("x-forwarded-proto", "").split(",")[0].strip().lower() == "https" 

34

36class SecurityHeadersMiddleware(BaseHTTPMiddleware): 

37 def __init__(self, app: Any, hsts: bool = True, csp_mode: str = "default") -> None: 

38 super().__init__(app) 

39 self.hsts = hsts 

40 self.csp_mode = csp_mode 

41 

42 async def dispatch(self, request: Request, call_next: Callable[[Request], Awaitable[Response]]) -> Response: 

43 response = await call_next(request) 

44 headers = response.headers 

45 headers.setdefault("X-Content-Type-Options", "nosniff") 

46 headers.setdefault("X-Frame-Options", "DENY") 

47 headers.setdefault("Referrer-Policy", "strict-origin-when-cross-origin") 

48 headers.setdefault("Permissions-Policy", "camera=(), microphone=(), geolocation=(), payment=(), usb=()") 

49 headers.setdefault("Cross-Origin-Opener-Policy", "same-origin") 

50 headers.setdefault("Cross-Origin-Resource-Policy", "same-origin") 

51 headers.setdefault("Cache-Control", "no-store") if request.url.path.startswith("/api") else None 

52 if self.csp_mode != "off": 

53 path = request.url.path 

54 policy = DOCS_CSP if path.startswith(DOCS_PATHS) else STRICT_CSP 

55 if is_https(request): 

56 # Only meaningful behind TLS; on plain HTTP it would break same-origin API calls. 

57 policy += "; upgrade-insecure-requests" 

58 headers.setdefault("Content-Security-Policy", policy) 

59 if self.hsts and is_https(request): 

60 headers.setdefault("Strict-Transport-Security", "max-age=63072000; includeSubDomains") 

61 return response

64class OriginCheckMiddleware(BaseHTTPMiddleware): 

65 """Reject state-changing requests that carry the session cookie but come from another origin. 

66 

67 Bearer-token requests are exempt (no ambient credential, no CSRF). SameSite=Lax on the cookie 

68 already blocks cross-site POSTs in modern browsers; this check is the defence in depth. 

69 """ 

70 

71 def __init__(self, app: Any, allowed_origins: list[str] | None = None) -> None: 

72 super().__init__(app) 

73 self.allowed_origins = {origin.rstrip("/") for origin in (allowed_origins or [])} 

74 

75 async def dispatch(self, request: Request, call_next: Callable[[Request], Awaitable[Response]]) -> Response: 

76 if ( 

77 request.method in UNSAFE_METHODS 

78 and request.url.path.startswith("/api") 

79 and SESSION_COOKIE in request.cookies 

80 and not request.headers.get("authorization") 

81 ): 

82 origin = request.headers.get("origin") or _origin_of(request.headers.get("referer")) 

83 if origin is None: 

84 return PlainTextResponse("missing Origin header on a cookie-authenticated request", status_code=403) 

85 if origin.rstrip("/") not in self.allowed_origins and _host_of(origin) != request.headers.get("host"): 

86 return PlainTextResponse("cross-origin request rejected", status_code=403) 

87 return await call_next(request) 

88 

89 

90def _origin_of(referer: str | None) -> str | None: 

91 if not referer: 

92 return None 

93 parts = urlsplit(referer) 

94 if not parts.scheme or not parts.netloc: 

95 return None 

96 return f"{parts.scheme}://{parts.netloc}" 

97 

98 

99def _host_of(origin: str) -> str: 

100 return urlsplit(origin).netloc
