16 "content": {"type": "string", "description": "Complete HCL file content"},

18 "required": ["path", "content"],

41def build_hcl_prompt_parts( 

42 provider: str, 

43 cim_json: str, 

44 documentation_context: str, 

45 validation_feedback: str, 

46 retry_count: int, 

47 required_tags: dict[str, str], 

48 source_vcenter: str, 

49 scope: str = SCOPE_ALL, 

50 compute_unit_ids: list[str] | None = None, 

51 existing_resources: list[str] | None = None, 

52) -> tuple[str, str]: 

53 """Return (stable_prefix, volatile_prompt). 

54 

55 The stable prefix (requirements, full CIM, documentation) is identical for every batch 

56 and every retry of a run, so the client marks it cacheable. The volatile part carries 

57 the batch scope, the attempt index and the validation feedback. 

58 """ 

59 root_dir = f"{provider}-migration" 

60 structure = ( 

61 "{root_dir}/main.tf\n" 

62 "{root_dir}/networking/vpc.tf\n" 

63 "{root_dir}/networking/subnets.tf\n" 

64 "{root_dir}/networking/security_groups.tf\n" 

65 "{root_dir}/placement/placement_groups.tf\n" 

66 "{root_dir}/compute/{compute_unit_name}.tf\n" 

67 "{root_dir}/storage/{compute_unit_name}_disks.tf" 

68 ).replace("{root_dir}", root_dir) 

69 

70 stable = ( 

71 "Generate production-ready Terraform HCL for the given CIM.\n\n" 

72 "Hard requirements:\n" 

73 '1. Return a JSON object {"files": [{"path": "...", "content": "..."}]} and nothing else.\n' 

74 "2. Paths must follow this structure:\n" 

75 f"{structure}\n" 

76 "3. Every resource must include tags: " 

77 f"{json.dumps(required_tags, ensure_ascii=True)}\n" 

78 "4. Never generate public S3 ACLs, never generate ingress cidr_blocks with 0.0.0.0/0.\n" 

79 "5. Never use default VPC; always define explicit network resources.\n" 

80 "6. Preserve 1:1 NetworkTopology mapping (DistributedSwitch to VPC/VNet/Network, PortGroup to subnet).\n" 

81 "7. If a ComputeUnit has has_vtpm=true, add TODO comment in HCL for manual review.\n" 

82 "8. Keep ComputeUnit vcpus and ram_mb unchanged from CIM; use only provided sizing decisions.\n" 

83 f"9. Source vCenter for tags/context: {source_vcenter}.\n" 

84 "10. Never emit provisioner blocks, external or http data sources, null_resource, local_file, " 

85 "remote module sources, or providers other than the target cloud's.\n" 

86 f"11. {root_dir}/main.tf must be a complete runnable root module for terraform init/validate/plan.\n" 

87 "12. Do not assume files in subdirectories are loaded together as one module.\n" 

88 "13. If using local.common_tags in a file, declare it in that same file or pass it via module inputs.\n" 

89 "14. Do not output informational/comment-only Terraform files for compute/networking when ComputeUnits are present.\n" 

90 "15. Generated HCL must include concrete resource blocks needed to run terraform plan.\n\n" 

91 f"{_provider_hcl_constraints(provider)}" 

92 "CIM JSON (complete inventory for context):\n" 

93 f"{cim_json}\n\n" 

94 "MCP Documentation Context:\n" 

95 f"{documentation_context}\n" 

96 ) 

97 

98 volatile = ( 

99 f"{_scope_instruction(scope, compute_unit_ids, existing_resources)}\n" 

100 f"This is retry attempt index: {retry_count}. Fix prior validation issues if provided.\n\n" 

101 "Validation Feedback:\n" 

102 f"{validation_feedback}\n" 

103 ) 

104 return stable, volatile

107def build_hcl_user_prompt( 

108 provider: str, 

109 cim_json: str, 

110 documentation_context: str, 

111 validation_feedback: str, 

112 retry_count: int, 

113 required_tags: dict[str, str], 

114 source_vcenter: str, 

115 scope: str = SCOPE_ALL, 

116 compute_unit_ids: list[str] | None = None, 

117 existing_resources: list[str] | None = None, 

118) -> str: 

119 """Single-string form of build_hcl_prompt_parts for clients without a cacheable prefix.""" 

120 stable, volatile = build_hcl_prompt_parts( 

121 provider=provider, 

122 cim_json=cim_json, 

123 documentation_context=documentation_context, 

124 validation_feedback=validation_feedback, 

125 retry_count=retry_count, 

126 required_tags=required_tags, 

127 source_vcenter=source_vcenter, 

128 scope=scope, 

129 compute_unit_ids=compute_unit_ids, 

130 existing_resources=existing_resources, 

131 ) 

132 return f"{stable}\n{volatile}"

135def _scope_instruction(scope: str, compute_unit_ids: list[str] | None, existing_resources: list[str] | None) -> str: 

136 if scope == SCOPE_SHARED: 

137 return ( 

138 "Scope for this response: ONLY the shared files - main.tf, networking/*.tf and placement/*.tf. " 

139 "Do not emit any compute/ or storage/ files." 

140 ) 

141 if scope == SCOPE_COMPUTE: 

142 ids = ", ".join(compute_unit_ids or []) 

143 existing = "\n".join(f"- {address}" for address in (existing_resources or [])) or "- (none)" 

144 return ( 

145 "Scope for this response: ONLY compute/ and storage/ files for these ComputeUnit ids: " 

146 f"{ids}. Do not emit main.tf, networking or placement files and do not redefine any of these " 

147 f"resources, which already exist and may be referenced by address:\n{existing}" 

148 ) 

149 return "Scope for this response: all files for the complete CIM."

154 if normalized_provider != "gcp":

160 "17. GCP label keys must be lowercase snake_case (for example environment, owner, migrated_from).\n\n"
